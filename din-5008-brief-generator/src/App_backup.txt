        const targetHeight = 1170;  // 1122.5px paper height + 48px safe padding
        
        if (containerWidth > 0 && containerHeight > 0) {
          const scaleX = containerWidth / targetWidth;

          setCanvasScale(1);

              if (p.signatureName !== undefined) setSignatureName(p.signatureName);
              if (p.attachments !== undefined) setAttachments(p.attachments);
              if (p.specialNotes !== undefined) setSpecialNotes(p.specialNotes);

  const fetchSharingCodeForUser = async (uid: string) => {
    try {
      const q = query(collection(db, 'sharingCodes'), where('uid', '==', uid));
      const snap = await getDocs(q);
      if (!snap.empty) {
        setSharingCode(snap.docs[0].id);
      } else {
        // Generate a cryptographically pleasant 6-character clean share code
        const safeChars = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789'; // zero-ambiguity characters
        const newCode = Array.from({ length: 6 }, () => 
          safeChars.charAt(Math.floor(Math.random() * safeChars.length))
        ).join('');
        await setDoc(doc(db, 'sharingCodes', newCode), {
          uid,
          createdAt: new Date().toISOString()
        });
        setSharingCode(newCode);
      }
    } catch (err) {
      console.error("Fehler bei der Generierung des Teilen-Codes:", err);
    }
  };

      setLoginPassword("");
    } catch (err: any) {
      setAuthError(err.message || "Registrierung fehlgeschlagen.");
    }
  };

      setAuthError("Bitte geben Sie E-Mail und Passwort ein.");
      return;
    }
    try {
      await signInWithEmailAndPassword(auth, loginEmail, loginPassword);
      alert("Erfolgreich angemeldet!");
      setLoginEmail("");
      setLoginPassword("");
    } catch (err: any) {
      setAuthError(err.message || "Anmeldung fehlgeschlagen.");
    }
  };

  const handleLogout = async () => {
    if (confirm("Möchten Sie sich wirklich abmelden?")) {
      try {
        await auth.signOut();

        setSharingCode("");

        console.error(err);

    if (code.length !== 6) {
      alert("Bitte geben Sie einen gültigen 6-stelligen Code ein.");
      return;
    }
    setIsSharingCodeLoading(true);
    try {
      const codeDocRef = doc(db, 'sharingCodes', code);

      if (!codeSnap.exists()) {
        alert("Dieser Code ist ungültig oder wurde bereits gelöscht.");
        setIsSharingCodeLoading(false);
        return;

      const targetUid = codeSnap.data().uid;
      const targetDocRef = doc(db, 'users', targetUid);
      const targetSnap = await getDoc(targetDocRef);

          if (data.settings.formality) setFormality(data.settings.formality);
          if (data.settings.dateFormat) setDateFormat(data.settings.dateFormat);
          if (data.settings.showGuides !== undefined) setShowGuides(data.settings.showGuides);
          if (data.settings.isDarkMode !== undefined) setIsDarkMode(data.settings.isDarkMode);

          setSenderName(data.senderProfile.senderName || "");
          setSenderStreet(data.senderProfile.senderStreet || "");
          setSenderZipCity(data.senderProfile.senderZipCity || "");

        script.onload = () => setGoogleStatus('ready');

        return formality === 'formal' ? "Sehr geehrte Damen und Herren," : "Hallo zusammen,";
      }

      const isFrau = /\b(Frau|Fr\.)\b/i.test(recipientName);
      const isHerr = /\b(Herr|Hr\.)\b/i.test(recipientName);

      // Filter out prefixes
      const cleanName = recipientName.replace(/\b(Frau|Herr|Fr\.|Hr\.|Dr\.|Prof\.)\b/gi, '').trim();

        if (isHerr) return `Sehr geehrter Herr ${lastName},`;

        if (isFrau) return `Guten Tag Frau ${lastName},`;
        if (isHerr) return `Guten Tag Herr ${lastName},`;
        return "Guten Tag,";
      } else {
        return `Hallo ${firstName || lastName},`;
      }

      document.documentElement.classList.add('dark');

                provider: 'google',
                label: p.description,
                placeId: p.place_id,
                mainText: p.structured_formatting?.main_text || "",
                secondaryText: p.structured_formatting?.secondary_text || ""

                provider: 'geoapify',
                label: f.properties.formatted,

              })));
            }

            `https://photon.komoot.io/api/?q=${encodeURIComponent(addressQuery)}&limit=5&lang=de`
