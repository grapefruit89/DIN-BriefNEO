export const EVENT_PREFIX = 'dinbrief:';

/**
 * Zentrale typisierte Event-Namen für das Pub/Sub-System.

export const Events = Object.freeze({
  NOTIFICATION_REQUESTED: 'dinbrief:notification.requested',
  DOCUMENT_SAVE_REQUESTED: 'dinbrief:document.save.requested',
  DOCUMENT_SAVED: 'dinbrief:document.saved',
  DOCUMENT_SAVE_FAILED: 'dinbrief:document.save.failed',

  DOCUMENT_CHANGED: 'dinbrief:document.changed'

/**
 * Erzeugt eine isolierte EventBus-Instanz.
 * @param {EventTarget} [target]
 */
export function createEventBus(target = new EventTarget()) {
  return Object.freeze({
    /**
     * Publiziert ein typisiertes Event an alle aktiven Subscriber.
     * @param {string} topic
     * @param {Record<string, unknown>} [payload]
     */
    publish(topic, payload = {}) {
      if (!topic.startsWith(EVENT_PREFIX)) {
        throw new TypeError(`Ungültiges Event-Topic: "${topic}". Topic muss mit "${EVENT_PREFIX}" beginnen.`);
      }

      target.dispatchEvent(
        new CustomEvent(topic, {
          detail: payload
        })
      );
    },

    /**
     * Registriert einen Handler für ein bestimmtes Topic.
     * @param {string} topic
     * @param {(detail: any) => void} handler
     * @param {{ signal?: AbortSignal }} [options]
     * @returns {() => void} Unsubscribe-Funktion
     */
    subscribe(topic, handler, options = {}) {
      if (!topic.startsWith(EVENT_PREFIX)) {
        throw new TypeError(`Ungültiges Event-Topic: "${topic}". Topic muss mit "${EVENT_PREFIX}" beginnen.`);
      }

      /**
       * @param {Event} event
       */
      const listener = (event) => {
        if (event instanceof CustomEvent) {
          handler(event.detail);
        }
      };

      target.addEventListener(topic, /** @type {EventListener} */ (listener), {
        signal: options.signal
      });

      return () => {
        target.removeEventListener(topic, /** @type {EventListener} */ (listener));
      };
    }
  });
}

# DIN-BriefNEO

> Normgerechter, datenschutzfreundlicher Brief-Editor nach DIN 5008 (Form A und Form B) für geschäftliche und private Korrespondenz.

## Übersicht

DIN-BriefNEO ist eine reine Client-Side-Webanwendung zur Erstellung und Formatierung von Briefen nach der deutschen Industrienorm DIN 5008.

- **Architektur:** 100 % clientseitig, Zero-Dependency, keine externen Frameworks, keine Cloud-Datenspeicherung.
- **Datenschutz:** Alle Daten verbleiben im Browser des Nutzers (LocalStorage für Entwürfe, In-Memory-Blobs für Signaturen). Eine Ausnahme: die optionale Adress-Suche sendet die Eingabe an `api.geoapify.com`, sobald ein API-Key hinterlegt ist; ohne Key findet kein externer Aufruf statt.
- **Export:** Nativer Browser-Druckdialog (Druck / PDF) mit exakter mm-Typografie via Container Queries und @page.
- **Datenformat:** Reines Text-JSON (Schema v5 Tuple-Format, ohne Base64).
- **Nicht unterstützt:** Teilbetreff (DIN 20.9.3) — 2026-11-11 bewusst entfernt. Der Unterrand ist optional auf 30 mm umstellbar; in dieser Einstellung ist der Brief nicht mehr DIN-konform.

## Dokumenten-Schema (Schema v5, Tuple-Format)

Das Dokumentenformat `.json` folgt derselben hocheffizienten Tabular-/Tuple-Architektur wie `figuren.json` (`_meta.schema` + Werte-Array `fields`), statt redundante Feldbezeichner als Key-Value-Blob mitzuschleifen. Es ist vollständig menschenlesbar und enthält keine Base64- oder Binärdaten. Signaturbild und eigene Schrift bleiben lokal (IndexedDB) und sind nicht Teil der Datei.
