/*
 * Blocking classic script (head): setzt Theme, Stil und Layout-Flags VOR dem
 * ersten Paint (FOUC-Schutz). Aus boot-theme.js und boot-switches.js
 * zusammengelegt (2026-11-08): beide lasen denselben localStorage-Eintrag,
 * der Browser blockierte den FCP zweimal und parste das JSON zweimal.
 *
 * Warum kein Modul: Module sind deferred und kaemen nach dem ersten Paint
 * (gemessen: FCP bei 96 ms, Zustand dort noch "aus").
 */
try {
  const defaults = {
    theme: 'auto',
    layout: 'form-b',
    style: 'default',
    marks: true,
    pluginTestbrief: true,
    randU: 20
  };

  let parsed = {};
  try {
    parsed = JSON.parse(localStorage.getItem('din_settings') || '{}');
  } catch {
    /* kaputtes JSON: mit Defaults weiter */
  }
  const settings = parsed && typeof parsed === 'object' ? { ...defaults, ...parsed } : defaults;
  const root = document.documentElement;

  /** @type {(v: any, def: boolean) => boolean} */
  const asBool = (v, def) => (v === undefined ? def : v === true || v === 'true');

  const THEME_WHITELIST = new Set(['auto', 'light', 'dark']);
  const LAYOUT_WHITELIST = new Set(['form-a', 'form-b']);

  const theme = THEME_WHITELIST.has(settings.theme) ? settings.theme : 'auto';
  const layout = LAYOUT_WHITELIST.has(settings.layout) ? settings.layout : 'form-b';

  root.setAttribute('data-theme', theme);
  root.setAttribute('data-layout', layout);
  root.setAttribute('data-style', settings.style || 'default');
  root.setAttribute('data-marks', asBool(settings.marks, true) ? 'true' : 'false');
  root.setAttribute('data-plugin-testbrief', asBool(settings.pluginTestbrief, true) ? 'true' : 'false');
  /* Der Einzelseiten-Modus ist stillgelegt ([[ADR-EINSEITEN-STILLGELEGT]]):
   * das Attribut bleibt fuer CSS/Tests gesetzt, ist aber immer 'true', und
   * es gibt keinen Stylesheet-Tausch mehr. `sheet.css` bleibt ungenutzt
   * liegen (nichts geloescht). */
  root.setAttribute('data-plugin-multipage', 'true');

  /* Unterrand 20/30mm vor dem ersten Paint setzen (FOUC-Schutz): die
   * Druckregel von #kern liest --rand-u. Nur 20 und 30 sind zulaessig. */
  const RAND_U_WHITELIST = new Set([20, 30]);
  const randU = RAND_U_WHITELIST.has(Number(settings.randU)) ? Number(settings.randU) : 20;
  root.style.setProperty('--rand-u', `${randU}mm`);

  /* Sicherheitsnetz fuer den Platzhalter-FOUC-Schutz (sheet-common.css):
   * Bleibt initApp haengen, duerfen die Platzhalter nicht dauerhaft
   * verschwinden. Nach 2 s wird freigegeben, komme was mag. */
  setTimeout(() => {
    if (!root.dataset.ready) root.dataset.ready = 'true';
  }, 2000);
} catch (e) {
  /* Ohne Boot-Zustand startet die App im falschen Farbschema — sichtbar,
   * aber ohne Ursache. Daher loggen. */
  console.error('[Boot] Initialisierung fehlgeschlagen:', e);
}

export const EVENT_PREFIX = 'dinbrief:';

/**
 * Zentrale typisierte Event-Namen für das Pub/Sub-System.
 */
export const Events = Object.freeze({
  NOTIFICATION_REQUESTED: 'dinbrief:notification.requested',
  DOCUMENT_SAVE_REQUESTED: 'dinbrief:document.save.requested',
  DOCUMENT_SAVED: 'dinbrief:document.saved',
  DOCUMENT_SAVE_FAILED: 'dinbrief:document.save.failed',
  PLUGIN_MULTIPAGE_TOGGLED: 'dinbrief:plugin.multipage.toggled',
  DOCUMENT_CHANGED: 'dinbrief:document.changed'
});

/**
 * Erzeugt eine isolierte EventBus-Instanz.
 * @param {EventTarget} [target]
 */
export function createEventBus(target = new EventTarget()) {
  return Object.freeze({
    /**
     * Publiziert ein typisiertes Event an alle aktiven Subscriber.
     * @param {string} topic
     * @param {Record<string, unknown>} [payload]
     */
    publish(topic, payload = {}) {
      if (!topic.startsWith(EVENT_PREFIX)) {
        throw new TypeError(`Ungültiges Event-Topic: "${topic}". Topic muss mit "${EVENT_PREFIX}" beginnen.`);
      }

      target.dispatchEvent(
        new CustomEvent(topic, {
          detail: payload
        })
      );
    },

    /**
     * Registriert einen Handler für ein bestimmtes Topic.
     * @param {string} topic
     * @param {(detail: any) => void} handler
     * @param {{ signal?: AbortSignal }} [options]
     * @returns {() => void} Unsubscribe-Funktion
     */
    subscribe(topic, handler, options = {}) {
      if (!topic.startsWith(EVENT_PREFIX)) {
        throw new TypeError(`Ungültiges Event-Topic: "${topic}". Topic muss mit "${EVENT_PREFIX}" beginnen.`);
      }

      /**
       * @param {Event} event
       */
      const listener = (event) => {
        if (event instanceof CustomEvent) {
          handler(event.detail);
        }
      };

      target.addEventListener(topic, /** @type {EventListener} */ (listener), {
        signal: options.signal
      });

      return () => {
        target.removeEventListener(topic, /** @type {EventListener} */ (listener));
      };
    }
  });
}

    }
  });

  eventBus.subscribe(Events.DOCUMENT_SAVE_FAILED, (payload) => {
    const msg = (payload && typeof payload.message === 'string') ? payload.message : Constants.TOASTS.SYSTEM.SAVE_ERROR;
    showToast(msg, 'error');
  });
}

/**
 * @param {string} message
 * @param {string} [type]
 */
export function showToast(message, type = 'info') {
  if (!dom) return;

  if (current?.message === message) {
    if (dom.global.dataset.type !== type) {
      dom.global.dataset.type = type;
    }
    showPopoverSafe(dom.global);
    startTimer(message);
    return;
  }

  dom.message.textContent = message;
  dom.global.dataset.type = type;
  current = { message };

  showPopoverSafe(dom.global);
  startTimer(message);
}

/** @param {string} message */
function startTimer(message) {
  if (timer !== null) clearTimeout(timer);
  const duration = Math.min(5000, 2000 + message.length * 30);
  timer = setTimeout(hide, duration);
}

function hide() {
  if (timer !== null) { clearTimeout(timer); timer = null; }
  current = null;
  hidePopoverSafe(dom?.global);
}

export function notify(message, type = 'info') {
  eventBus.publish(Events.NOTIFICATION_REQUESTED, { message, type });
}

# DIN-BriefNEO

> Normgerechter, datenschutzfreundlicher Brief-Editor nach DIN 5008 (Form A und Form B) für geschäftliche und private Korrespondenz.

## Übersicht

DIN-BriefNEO ist eine reine Client-Side-Webanwendung zur Erstellung und Formatierung von Briefen nach der deutschen Industrienorm DIN 5008.

- **Architektur:** 100 % clientseitig, Zero-Dependency, keine externen Frameworks, keine Cloud-Datenspeicherung.
- **Datenschutz:** Alle Daten verbleiben im Browser des Nutzers (LocalStorage für Entwürfe, In-Memory-Blobs für Signaturen). Eine Ausnahme: die optionale Adress-Suche sendet die Eingabe an `api.geoapify.com`, sobald ein API-Key hinterlegt ist; ohne Key findet kein externer Aufruf statt.
- **Export:** Nativer Browser-Druckdialog (Druck / PDF) mit exakter mm-Typografie via Container Queries und @page.
- **Datenformat:** Reines Text-JSON (Schema v5 Tuple-Format, ohne Base64).
- **Nicht unterstützt:** Teilbetreff (DIN 20.9.3) — 2026-11-11 bewusst entfernt. Der Unterrand ist optional auf 30 mm umstellbar; in dieser Einstellung ist der Brief nicht mehr DIN-konform.

## Dokumenten-Schema (Schema v5, Tuple-Format)

Das Dokumentenformat `.json` folgt derselben hocheffizienten Tabular-/Tuple-Architektur wie `figuren.json` (`_meta.schema` + Werte-Array `fields`), statt redundante Feldbezeichner als Key-Value-Blob mitzuschleifen. Es ist vollständig menschenlesbar und enthält keine Base64- oder Binärdaten. Signaturbild und eigene Schrift bleiben lokal (IndexedDB) und sind nicht Teil der Datei. Die App ist reines Client-JavaScript ohne Build-Schritt und ohne Server: `website/` auf einen beliebigen statischen Webserver kopieren genügt.

## DIN 5008 Geometrie & Normvorgaben

- **Form A:** Hoher Briefkopf (Anschriftfeld-Oberkante 34 mm, Empfängeranschrift 46,7 mm, Falzmarke 1 bei 87 mm).
- **Form B:** Tiefer Briefkopf (Anschriftfeld-Oberkante 51 mm, Empfängeranschrift 63,7 mm, Falzmarke 1 bei 105 mm). Standard für Fensterbriefumschläge DIN lang (DL).
- **Fluchtlinie:** Linker Blattrand 25 mm für alle Anschrift- und Textfelder.
