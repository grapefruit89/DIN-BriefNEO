// @ts-check
// @adr [[ADR-BINARY-ASSETS]] {openDB, saveBlob, loadBlob, deleteBlob}
// @guide [[glossary]]

/**
 * @file binary-storage.js — Binär-Assets (Signaturbild, Schriftdatei) in IndexedDB.
 *
 * `localStorage` kann nur Strings speichern; Base64 bläht Binärdaten um ~33 % auf und ist
 * im Quota teuer. Deshalb liegen die beiden Binär-Assets als `Blob` in IndexedDB.
 * **Strukturierte** Produktdaten (Draft, Einstellungen, Adressen, Schema-Version) bleiben
 * in `localStorage` — synchron und ohne FOUC (Law S1, A34).
 *
 * Einziger Speicherort für Blobs. `tools/antipatterns/project.json` (P10) exempted genau
 * diese Datei, damit IndexedDB nicht versehentlich als Produktspeicher zurückkehrt.
 */

/* Kein `din-`-Praefix im DB-Namen: dieser Namensraum ist der IMR-Registry vorbehalten
 * und wird von tools/imr.js als Tag gelesen. */
const DB_NAME = 'briefneo-assets';
const DB_VERSION = 1;
const STORE = 'assets';

/** @type {Promise<IDBDatabase> | null} */
let dbPromise = null;

/**
 * Öffnet die Datenbank lazy — erst beim ersten Zugriff. Ein Fehlschlag wird nicht
 * dauerhaft zwischengespeichert, damit ein späterer Versuch es erneut probieren kann.
 * @returns {Promise<IDBDatabase>}
 */
function openDB() {
  if (dbPromise) return dbPromise;
  if (typeof indexedDB === 'undefined') {
    return Promise.reject(new Error('IndexedDB nicht verfügbar'));
  }
  const { promise, resolve, reject } = Promise.withResolvers();
  const request = indexedDB.open(DB_NAME, DB_VERSION);
  request.onupgradeneeded = () => {
    const db = request.result;
    if (!db.objectStoreNames.contains(STORE)) db.createObjectStore(STORE);
  };
  request.onsuccess = () => {
    const db = request.result;
    db.onversionchange = () => {
      db.close();
      dbPromise = null;
    };
    resolve(db);
  };
  request.onerror = () => reject(request.error || new Error('IndexedDB open fehlgeschlagen'));
  request.onblocked = () => reject(new Error('IndexedDB blockiert (anderer Tab hält die alte Version)'));
  dbPromise = promise;
  promise.catch(() => {
    if (dbPromise === promise) dbPromise = null;
  });
  return promise;
}

/**
 * Speichert einen Blob unter einem Key. Schlägt der Speicher fehl (privater Modus,
 * deaktivierte Speicher-API), bleibt es bei einer Warnung — die Oberfläche meldet den
 * Fehlschlag selbst.
 * @param {string} key — 'signature' oder 'font'
 * @param {Blob} blob
 * @returns {Promise<boolean>} true bei Erfolg
 */
export async function saveBlob(key, blob) {
  if (!(blob instanceof Blob)) return false;
  try {
    const db = await openDB();
    const { promise, resolve, reject } = Promise.withResolvers();
    const tx = db.transaction(STORE, 'readwrite');
    tx.objectStore(STORE).put(blob, key);
    tx.oncomplete = () => resolve(true);
    tx.onerror = () => reject(tx.error || new Error('Transaktion fehlgeschlagen'));
    tx.onabort = () => reject(tx.error || new Error('Transaktion abgebrochen'));
    await promise;
    return true;
  } catch (error) {
    console.warn('[BinaryStorage] Speichern fehlgeschlagen, nichts gespeichert:', error);
    return false;
  }
}

/**
 * Lädt einen Blob. Fehlt der Key oder ist der Speicher nicht verfügbar, kommt `null`.
 * @param {string} key
 * @returns {Promise<Blob | null>}
 */
export async function loadBlob(key) {
  try {
    const db = await openDB();
    const { promise, resolve, reject } = Promise.withResolvers();
    const tx = db.transaction(STORE, 'readonly');
    const request = tx.objectStore(STORE).get(key);
    request.onsuccess = () => {
      const value = request.result;
      resolve(value instanceof Blob ? value : null);
    };
    request.onerror = () => reject(request.error || new Error('Lesen fehlgeschlagen'));
    tx.onabort = () => reject(tx.error || new Error('Transaktion abgebrochen'));
    return await promise;
  } catch (error) {
    console.warn('[BinaryStorage] Laden fehlgeschlagen:', error);
    return null;
  }
}

/**
 * Löscht einen Blob. Ein fehlender Key ist kein Fehler.
 * @param {string} key
 * @returns {Promise<boolean>}
 */
export async function deleteBlob(key) {
  try {
    const db = await openDB();
    const { promise, resolve, reject } = Promise.withResolvers();
    const tx = db.transaction(STORE, 'readwrite');
    tx.objectStore(STORE).delete(key);
    tx.oncomplete = () => resolve(true);
    tx.onerror = () => reject(tx.error || new Error('Löschen fehlgeschlagen'));
    tx.onabort = () => reject(tx.error || new Error('Löschen abgebrochen'));
    await promise;
    return true;
  } catch (error) {
    console.warn('[BinaryStorage] Löschen fehlgeschlagen:', error);
    return false;
  }
}

// @ts-check
// @adr [[ADR-DATA-PERSISTENCE]] {DraftManager}

/**
 * @file draft-manager.js — Autosave, Undo/Redo, Mehr-Tab-Sync und der
 * Speicherzustand des Entwurfs. Aus `document.js` ausgelagert (2026-11-08):
 * die State-Machine hat nichts mit DOM- und Caret-Manipulation zu tun.
 */

import { AddressIntelligence } from './address.js';
import { Constants, Events, eventBus } from './core.js';
import { collectDraft, EDITABLE_FIELD_SELECTOR, getCaretCharacterOffset, sanitizeRichText, setCaretCharacterOffset } from './document.js';
import { Field } from './field.js';
import { isPrintTitleActive, loadJSON, remove, STORAGE_KEYS, save, saveJSON, stripHtmlToPlaintext } from './persistence.js';

export class DraftManager {
  /** @type {Array<{draftStr: string, caretInfo: {id: string, offset: number} | null}>} */
  #undoStack = [];
  /** @type {Array<{draftStr: string, caretInfo: {id: string, offset: number} | null}>} */
  #redoStack = [];
  /** @type {{draftStr: string, caretInfo: {id: string, offset: number} | null} | null} */
  #currentState = null;
  /** @type {boolean} */
  #isRestoring = false;
  /** @type {(() => void) | null} */
  #unsubSaveRequested = null;
  /** @type {(() => void) | null} */
  #unsubSaveFailed = null;
  /**
   * Deterministischer Speicherzustand:
   * pending  = Änderungen im DOM, noch nicht geschrieben
   * synced   = letzter Schreibvorgang erfolgreich
   * error    = letzter Schreibvorgang fehlgeschlagen (Quota/Serde)
   * @type {'pending' | 'synced' | 'error'}
   */
  #saveState = 'synced';

  /**
   * @param {ReturnType<typeof import('./core.js').createEventBus> | null} [bus]
   */
  constructor(bus = eventBus) {
    /** @type {ReturnType<typeof import('./core.js').createEventBus> | null} */
    this.bus = bus;
    /** @type {any} */
    this.debounceTimer = null;
    this.DEBOUNCE_DELAY = 500;
    this.#initShortcuts();
    this.#initEventBus();
    this.#initLifecycleFlush();
    this.#initMultiTabSync();
  }

  #initEventBus() {
    if (this.bus && typeof this.bus.subscribe === 'function') {
      this.#unsubSaveRequested = this.bus.subscribe(Events.DOCUMENT_SAVE_REQUESTED, (payload) => {
        if (payload?.debounce) {
          this.scheduleAutoSave();
        } else {
          this.saveDraft();
        }
      });
      // Auch Fehler aus anderen Speicherpfaden (z. B. Einstellungen/Signatur)
      // müssen am Status-Punkt sichtbar werden, nicht nur als Toast.
      this.#unsubSaveFailed = this.bus.subscribe(Events.DOCUMENT_SAVE_FAILED, () => {
        this.#setSaveStatus('error');
      });
    }
  }

  #initLifecycleFlush() {
    const flush = () => this.flushPendingSave();

    if (typeof document !== 'undefined') {
      document.addEventListener('visibilitychange', () => {
        if (document.visibilityState === 'hidden') flush();
      });
    }

    if (typeof window !== 'undefined') {
      window.addEventListener('pagehide', flush);
    }
  }

  /** Aktueller Speicherzustand (pending | synced | error). */
  get saveState() {
    return this.#saveState;
  }

  /**
   * Schreibt anstehende Änderungen sofort. No-op, wenn nichts offen ist —
   * verhindert zusätzliche Schreibvorgänge beim bloßen Tab-Wechsel.
   * @returns {boolean} true, wenn geschrieben wurde
   */
  flushPendingSave() {
    if (this.debounceTimer === null && this.#saveState === 'synced') return false;
    if (this.debounceTimer !== null) {
      clearTimeout(this.debounceTimer);
      this.debounceTimer = null;
    }
    this.saveDraft();
    return true;
  }

  /* Bewusste Entscheidung: Kein Web-Locks-Mechanismus (Over-Engineering für
   * Single-User-App). Bei fokussiertem Tab gewinnt der aktive Tab; der Toast
   * warnt vor dem Multi-Tab-Fall. Restrisiko: paralleles Tippen in 2 Tabs
   * kann den älteren Stand überschreiben — akzeptiert. */
  #initMultiTabSync() {
    if (typeof window !== 'undefined') {
      window.addEventListener('storage', (event) => {
        if (event.key !== STORAGE_KEYS.draft || !event.newValue) return;
        // Wenn dieser Tab im Hintergrund ist und keine ungespeicherten Eingaben hat:
        if (typeof document !== 'undefined' && !document.hasFocus() && this.debounceTimer === null) {
          this.loadDraft();
        } else {
          eventBus.publish(Events.NOTIFICATION_REQUESTED, {
            message: Constants.TOASTS.DRAFT.MULTI_TAB,
            type: 'info'
          });
        }
      });
    }
  }

  destroy() {
    clearTimeout(this.debounceTimer);
    this.debounceTimer = null;
    if (this.#unsubSaveRequested) {
      this.#unsubSaveRequested();
      this.#unsubSaveRequested = null;
    }
    if (this.#unsubSaveFailed) {
      this.#unsubSaveFailed();
      this.#unsubSaveFailed = null;
    }
  }

  #initShortcuts() {
    // Shortcuts nur innerhalb des Briefblatts abfangen; andere Inputs behalten natives Undo/Redo
    document.addEventListener('keydown', (e) => {
      const blatt = /** @type {Element | null} */ (e.target instanceof Element ? e.target.closest('din-blatt') : null);
      if (!blatt || !(e.ctrlKey || e.metaKey)) return;
      const key = e.key.toLowerCase();
      if (key === 'z') {
        e.preventDefault();
        if (e.shiftKey) this.redo(); else this.undo();
      } else if (key === 'y') {
        e.preventDefault();
        this.redo();
      }
    });
  }

  saveDraft() {
    /* Ein expliziter Save macht einen geplanten Debounce-Save gegenstandslos:
     * sonst schreibt der Timer kurz danach denselben Stand ein zweites Mal. */
    if (this.debounceTimer !== null) {
      clearTimeout(this.debounceTimer);
      this.debounceTimer = null;
    }
    const draft = collectDraft();
    let draftStr = '';
    try {
      draftStr = JSON.stringify(draft);
    } catch {
      draftStr = '{}';
    }

    const saved = save(STORAGE_KEYS.draft, draftStr);
    this.#saveState = saved ? 'synced' : 'error';

    try {
      this.#setSaveStatus(saved ? 'saved' : 'error');
    } catch {
      /* Status-Rendering darf den Fehlerpfad niemals abbrechen (B54) */
    }

    try {
      this.#updateDocumentTitle();
    } catch {
      /* Titel-Update darf den Fehlerpfad nicht unterbrechen */
    }

    if (this.bus && typeof this.bus.publish === 'function') {
      try {
        this.bus.publish(saved ? Events.DOCUMENT_SAVED : Events.DOCUMENT_SAVE_FAILED, { draft });
      } catch (err) {
        console.warn('[DraftManager] Event-Publish fehlgeschlagen:', err);
      }
    }

    if (this.#isRestoring) return saved;

    let caretInfo = null;

    const activeElem = document.activeElement;
    if (activeElem?.hasAttribute('contenteditable') && activeElem.id && activeElem.id !== 'datum') {
      caretInfo = { id: activeElem.id, offset: getCaretCharacterOffset(activeElem) };
    }

    if (!this.#currentState) {
      this.#currentState = { draftStr, caretInfo };
    } else if (draftStr !== this.#currentState.draftStr) {
      this.#undoStack.push(this.#currentState);
      if (this.#undoStack.length > Constants.LIMITS.HISTORY_MAX_ITEMS) this.#undoStack.shift();
      this.#currentState = { draftStr, caretInfo };
      this.#redoStack = [];
    } else {
      this.#currentState.caretInfo = caretInfo;
    }

    return saved;
  }

  loadDraft() {
    const draft = loadJSON(STORAGE_KEYS.draft, null);
    if (!draft) return false;

    this.#currentState = { draftStr: JSON.stringify(draft), caretInfo: null };
    this.#restoreState(draft);
    return true;
  }

  /**
   * Stellt einen Snapshot (z. B. aus Undo nach Reset) im DOM wieder her und persistiert ihn.
   * @param {Record<string, string | boolean>} draft
   */
  restoreDraft(draft) {
    if (!draft || typeof draft !== 'object') return;
    this.#restoreState(draft);
    this.saveDraft();
  }

  /**
   * @param {Record<string, string | boolean>} draft
   */
  #restoreState(draft) {
    this.#isRestoring = true;
    AddressIntelligence.targetLock = null;
    Object.keys(draft).forEach(id => {
      const elem = document.getElementById(id);
      /* Schutz vor UI-Zerstörung durch fremde/manipulierte IDs: nur editierbare Felder beschreiben */
      if (!elem?.matches(EDITABLE_FIELD_SELECTOR)) return;
      if (elem.dataset.feldtyp === 'systemwert' && elem.id !== 'datum') return;
      // Boolean-Werte sind Toggle-Zustaende; sie werden weiter unten als `checked` gesetzt.
      if (typeof draft[id] !== 'string') return;

      if (elem instanceof HTMLSelectElement) {
        elem.value = draft[id];
        return;
      }

      if (Field.isNative(elem)) {
        /* Migration alter Drafts: <li>-HTML -> \n-Text (einmalig). */
        let val = draft[id];
        if (elem.id === 'anlagen-angabe' && typeof val === 'string' && val.includes('<')) {
          val = stripHtmlToPlaintext(val);
        }
        Field.setValue(elem, val);
      // HTML-Restore ausschließlich über sanitizeRichText (Allowlist)
      } else if (elem.getAttribute('data-feldtyp')?.includes('rich')) {
        const isListe = Boolean(elem.getAttribute('data-feldtyp')?.includes('liste'));
        if (isListe && typeof draft[id] === 'string' && !draft[id].includes('<li')) {
          const lines = draft[id].split('\n').map((l) => l.trim()).filter(Boolean);
          if (lines.length > 0) {
            elem.replaceChildren(...lines.map((line) => {
              const li = document.createElement('li');
              Field.setValue(li, line);
              return li;
            }));
          } else {
            elem.replaceChildren();
          }
        } else {
          const extra = isListe ? { extraTags: ['UL', 'OL', 'LI'] } : undefined;
          elem.replaceChildren(sanitizeRichText(draft[id], extra));
          if (isListe && elem.children.length === 0) {
            elem.replaceChildren(document.createElement('li'));
          }
        }
      } else if (!elem.querySelector('select[data-speichern]')) {
        Field.setValue(elem, draft[id]);
        if (elem.id === 'datum') {
          // Gemerkt: der Wert stammt vom Nutzer, applyLetterDate darf ihn nicht ueberschreiben.
          if ((draft[id] || '').trim() !== '') elem.dataset.dirty = 'true';
        }
      }
    });

    // Die Sichtbarkeit des Vermerks ist abgeleitet, kein gespeichertes Feld mehr:
    // leerer Vermerk = unsichtbar, gefuellter Vermerk = sichtbar.
    const postvermerk = document.getElementById('vermerke');
    if (postvermerk) {
      if (Field.getValue(postvermerk).trim() !== '') {
        postvermerk.setAttribute('data-visible', 'true');
      } else {
        postvermerk.removeAttribute('data-visible');
      }
    }
    const pvSelect = /** @type {HTMLSelectElement | null} */ (document.getElementById('seitenleiste-postvermerk-select'));
    if (pvSelect) {
      // Auswahlliste folgt dem Feld — sonst zeigt sie nach einem Restore „kein“,
      // während der Vermerk im Brief steht.
      pvSelect.value = Field.getValue(postvermerk).trim();
      if (pvSelect.selectedIndex === -1) pvSelect.selectedIndex = 0;
    }

    const anlagenSwitch = /** @type {HTMLInputElement | null} */ (document.getElementById('toggle-anlagen'));
    if (anlagenSwitch && 'toggle-anlagen' in draft) {
      anlagenSwitch.checked = draft['toggle-anlagen'] === true || draft['toggle-anlagen'] === 'true';
    }

    const fussSwitch = /** @type {HTMLInputElement | null} */ (document.getElementById('toggle-fuss'));
    if (fussSwitch) {
      if ('toggle-fuss' in draft) {
        fussSwitch.checked = draft['toggle-fuss'] === true || draft['toggle-fuss'] === 'true';
      } else {
        // Migration: alte Drafts kennen den Toggle nicht. Vorhandener Text
        // bedeutet "Fusszeile war an" - sonst waere er unsichtbar.
        fussSwitch.checked = Array.from(
          document.querySelectorAll('#geschaeftsangaben > [contenteditable]')
        ).some((sp) => Field.getValue(sp).trim() !== '');
      }
    }

    this.#isRestoring = false;
    this.#setSaveStatus('saved');
  }

  undo() {
    clearTimeout(this.debounceTimer);
    this.debounceTimer = null;
    this.saveDraft();
    if (this.#undoStack.length === 0 || !this.#currentState) return;
    this.#redoStack.push(this.#currentState);
    this.#currentState = this.#undoStack.pop() || null;
    this.#applyHistoryState(this.#currentState);
  }

  redo() {
    clearTimeout(this.debounceTimer);
    this.debounceTimer = null;
    this.saveDraft();
    if (this.#redoStack.length === 0 || !this.#currentState) return;
    this.#undoStack.push(this.#currentState);
    this.#currentState = this.#redoStack.pop() || null;
    this.#applyHistoryState(this.#currentState);
  }

  /**
   * @param {{draftStr: string, caretInfo: {id: string, offset: number} | null} | null} stateObj
   */
  #applyHistoryState(stateObj) {
    if (!stateObj) return;
    const draft = JSON.parse(stateObj.draftStr);
    this.#restoreState(draft);
    saveJSON(STORAGE_KEYS.draft, draft);

    if (stateObj.caretInfo) {
      const elem = document.getElementById(stateObj.caretInfo.id);
      if (elem) {
        elem.focus();
        setCaretCharacterOffset(elem, stateObj.caretInfo.offset);
      }
    }

    if (this.bus && typeof this.bus.publish === 'function') {
      this.bus.publish(Events.DOCUMENT_CHANGED, { draft });
    }
  }

  resetDraft() {
    document.querySelectorAll(EDITABLE_FIELD_SELECTOR).forEach(el => {
      if (el.getAttribute('data-feldtyp') === 'systemwert') return;
      if (Field.isNative(el)) {
        Field.setValue(el, '');
      } else if (el.getAttribute('data-feldtyp')?.includes('liste')) {
        if (el.id === 'anlagen-angabe') {
          el.replaceChildren();
        } else {
          el.replaceChildren(document.createElement('li'));
        }
      } else {
        el.replaceChildren();
        Field.setValue(el, '');
      }
    });
    document.querySelectorAll('select[data-speichern]').forEach(el => {
      const sel = /** @type {HTMLSelectElement} */ (el);
      sel.selectedIndex = 0;
    });
    const anlagenSwitch = /** @type {HTMLInputElement | null} */ (document.getElementById('toggle-anlagen'));
    if (anlagenSwitch) {
      anlagenSwitch.checked = false;
    }
    const fussSwitch = /** @type {HTMLInputElement | null} */ (document.getElementById('toggle-fuss'));
    if (fussSwitch) {
      fussSwitch.checked = false;
    }
    // Datum: Markierung loeschen und Feld leeren, damit applyLetterDate wieder "heute" setzt.
    const datumEl = document.getElementById('datum');
    if (datumEl) {
      datumEl.removeAttribute('data-dirty');
      Field.setValue(datumEl, '');
    }
    const postvermerk = document.getElementById('vermerke');
    if (postvermerk) {
      postvermerk.removeAttribute('data-visible');
    }
    const pvSelect = /** @type {HTMLSelectElement | null} */ (document.getElementById('seitenleiste-postvermerk-select'));
    if (pvSelect) {
      pvSelect.selectedIndex = 0;
    }
    AddressIntelligence.targetLock = null;
    /* Privacy: die lokale Adress-Historie haelt bis zu 50 zuletzt gewaehlte
     * Empfaengeradressen und gehoert damit zum Brief, nicht zu den
     * Praeferenzen. Der Reset muss sie mitnehmen, sonst bleiben echte
     * Adressen auf einem geteilten Rechner stehen (gemessen 2026-11-11).
     * din_settings/din_geoapify_key/din_sender_coords bleiben bewusst.
     * `remove()` faengt einen gesperrten Storage selbst ab. */
    remove(STORAGE_KEYS.addresses);
    /* Erst publizieren, dann speichern: Listener (Anrede- und Gruss-Seed)
     * schreiben ihren Folgezustand ins DOM, danach wird der fertige Stand in
     * EINEM Zug gesichert. Vorher entstanden zwei Schreibvorgaenge: leer, dann
     * geseedet. Der Publish steht im try/catch, damit ein fehlerhafter Listener
     * das Zuruecksetzen nicht unspeicherbar macht. */
    try {
      if (this.bus && typeof this.bus.publish === 'function') {
        this.bus.publish(Events.DOCUMENT_CHANGED, { draft: {} });
      }
    } catch (err) {
      console.warn('[DraftManager] DOCUMENT_CHANGED nach Reset fehlgeschlagen:', err);
    }
    this.saveDraft();
  }

  scheduleAutoSave() {
    this.#saveState = 'pending';
    this.#setSaveStatus('dirty');
    clearTimeout(this.debounceTimer);
    this.debounceTimer = setTimeout(() => {
      this.debounceTimer = null;
      this.saveDraft();
    }, this.DEBOUNCE_DELAY);
  }

  /**
   * Autosave-Indikator: Status-Dot im Sidebar-Footer.
   * Zustandsgetrieben über data-state="saved" | "dirty" | "error".
   * @param {'saved' | 'dirty' | 'error'} state
   */
  #setSaveStatus(state) {
    const el = /** @type {HTMLElement | null} */ (document.getElementById('save-status'));
    if (!el) return;
    el.dataset.state = state;
    const labels = { saved: 'Gespeichert', dirty: 'Speichern…', error: 'Fehler beim Speichern!' };
    Field.setValue(el, labels[state]);
    el.title = labels[state];

    // Fallback auf ARIA-Live-Regionen (B54: Screenreader auch ohne ariaNotify informieren)
    if (state === 'error') {
      el.setAttribute('role', 'alert');
      el.setAttribute('aria-live', 'assertive');
    } else {
      /* Im Regelbetrieb keine Live-Region: sonst kuendigt jeder Autosave
       * (~500 ms) "Speichern…" und "Gespeichert" an. Nur Fehler werden
       * angekuendigt — der Status bleibt visuell unveraendert. */
      el.removeAttribute('role');
      el.removeAttribute('aria-live');
    }

    if (state === 'error') {
      try {
        if (typeof document.ariaNotify === 'function') {
          // Standard-konforme Priorität 'high' oder 'normal' (Chrome 156+, B54)
          document.ariaNotify('Fehler beim Speichern', { priority: 'high' });
        }
      } catch {
        /* Experimentelle API darf den Fehlerpfad niemals blockieren (B54) */
      }
    }
  }

  #updateDocumentTitle() {
    if (isPrintTitleActive()) return;
    const betreff = (document.getElementById('betreff')?.textContent || '').replace(/[\x00-\x1f\r\n]/g, ' ').replace(/\s+/g, ' ').trim() || 'Unbenannt';
    document.title = betreff;
  }
}

/**
 * Liest ein JSON-Objekt aus dem LocalStorage.
 * @template T
 * @param {string} key
 * @param {T} fallback
 * @returns {T}
 */
export function loadJSON(key, fallback) {
  const raw = load(key);
  if (raw === null) return fallback;
  try {
    const parsed = JSON.parse(raw);
    if (parsed === null || typeof parsed !== typeof fallback || Array.isArray(parsed) !== Array.isArray(fallback)) {
      return fallback;
    }
    if (typeof fallback === 'object' && !Array.isArray(fallback)) {
      const clean = Object.create(null);
      for (const [k, v] of Object.entries(parsed)) {
        if (k !== '__proto__' && k !== 'constructor' && k !== 'prototype') {
          clean[k] = v;
        }
      }
      return /** @type {T} */ ({ ...fallback, ...clean });
    }
    return /** @type {T} */ (parsed);
  } catch {
    return fallback;
  }
}

/**
 * Kontext-Stack für reentrante beforeprint/afterprint-Zyklen.
 * Verhindert Deadlocks und Metadaten-Leaks bei asynchronen oder mehrfachen beforeprint-Events.
 * @type {Array<{ oldTitle: string, injectedTags: HTMLMetaElement[] }>}
 */
const printContextStack = [];

/**

export function isPrintTitleActive() {

export const MetadataService = {
  /**
   * Bereitet Titel und Meta-Tags vor dem Drucken vor.
   * Reentrant: Push auf den Kontext-Stack; ein verschachteltes beforeprint erbt
   * den echten Originaltitel des äußersten Aufrufs.
   */
  prepare() {
    this._cleanupInjectedTags();

    // Originaltitel: Wenn der Stack leer ist, ist document.title der echte Originaltitel.
    // Bei verschachteltem Aufruf erbt der neue Kontext den ursprünglichen Titel des ersten Eintrags.
    const currentTitle = typeof document !== 'undefined' ? document.title : '';
    const originalTitle = printContextStack.length > 0
      ? printContextStack[0].oldTitle
      : currentTitle;

    const { fileName, dateStr, lastName, recipientName, subjectClean } = collectLetterIdentity();
    if (typeof document !== 'undefined') {
      document.title = fileName;
    }

    const metaData = {
      author: lastName,
      description: `DIN 5008 Brief an ${recipientName} - ${subjectClean}`,
      keywords: `DIN-Brief, BriefNEO, ${recipientName}, ${dateStr}, Platinum`,
      title: fileName
    };

    const injectedTags = this._injectMetaTags(metaData);
    const ctx = { oldTitle: originalTitle, injectedTags };
    printContextStack.push(ctx);

    return ctx;
  },

  /**
   * Entfernt alle im Head injizierten Print-Metatags.
   */
  _cleanupInjectedTags() {
    if (typeof document === 'undefined') return;
    const existing = document.head.querySelectorAll('meta[data-injected="true"]');
    for (const tag of existing) {
      tag.remove();
    }
  },

  /**
   * @param {{ author: string, description: string, keywords: string, title: string }} data
   * @returns {HTMLMetaElement[]}
   */
  _injectMetaTags(data) {
    if (typeof document === 'undefined') return [];
    /** @type {HTMLMetaElement[]} */
    const tags = [];
    const mapping = {
      'author': data.author,
      'description': data.description,
      'keywords': data.keywords,
      'application-name': 'DIN-BriefNEO Platinum V5'
    };

    Object.entries(mapping).forEach(([name, content]) => {
      const meta = document.createElement('meta');
      meta.name = name;
      meta.content = content;
      meta.setAttribute('data-injected', 'true');
      document.head.appendChild(meta);
      tags.push(meta);
    });
    return tags;
  },

  /**
   * Stellt Originaltitel wieder her und räumt injizierte Meta-Tags auf.
   * @param {{ oldTitle?: string, injectedTags?: HTMLMetaElement[] } | null} [context]
   */
  restore(context = null) {
    if (context && printContextStack.includes(/** @type {any} */ (context))) {
      const idx = printContextStack.indexOf(/** @type {any} */ (context));
      printContextStack.splice(idx, 1);
    } else if (printContextStack.length > 0) {
      printContextStack.pop();
    }

    this._cleanupInjectedTags();

    // Wenn der Stack wieder komplett leer ist, stellen wir den echten Originaltitel wieder her
    if (printContextStack.length === 0) {
      const restoreTitle = context?.oldTitle;
      if (restoreTitle && typeof document !== 'undefined') {
        document.title = restoreTitle;
      }
    }
  }
};

/**
 * Bereinigt HTML-Strings (wie <li> in Listenfeldern) zu lesbarem Plaintext für den JSON-Export.
 * @param {string} html
 * @returns {string}
 */
export function stripHtmlToPlaintext(html) {
  if (!html?.includes('<')) return html ? html.trim() : '';
  let str = html
    .replace(/<li[^>]*>/gi, '')
    .replace(/<\/li>/gi, '\n')
    .replace(/<br\s*\/?>/gi, '\n');
  let prev;
  do {
    prev = str;
    str = str.replace(/<[^>]+>/g, '');
  } while (str !== prev);
  return str
    .split('\n')
    .map((line) => line.trim())
    .filter(Boolean)
    .join('\n');
}

/**
 * Baut das Export-Payload (Schema v4, Tuple-Format) aus einem Draft-Objekt und den Einstellungen.
 * Folgt derselben hocheffizienten Tabular-/Tuple-Architektur wie figuren.json (_meta.schema + Werte-Array),
 * statt redundante Feldbezeichner als Key-Value-Blob mitzuschleifen.
 * Nichtbinäre Anteile (Signaturbild, eigene Schrift) sind bewusst nicht enthalten —
 * sie gehören in den separaten Container. PURE — unit-testbar.
 * @param {Record<string, string | boolean>} draft
 * @param {Record<string, any> | null} [settings]
 * @returns {DinLetterPayload}
 */
export function buildDinLetterPayload(draft, settings = null) {
  /** @type {string} */
  let generatedAt;
  try {
    generatedAt = Temporal.Now.zonedDateTimeISO('Europe/Berlin').toString();
  } catch {
    generatedAt = `${currentISODate()}T00:00:00+01:00`;
  }

  const draftObj = (draft && typeof draft === 'object') ? draft : {};

  // Extrahiere Werte in der kanonischen Reihenfolge von DOCUMENT_SCHEMA_V5
  /** @type {(string | boolean)[]} */
  const fields = DOCUMENT_SCHEMA_V5.map((field) => {
    if (field.startsWith('toggle-')) {
      const val = draftObj[field];
      return val === true || val === 'true';
    }
    const val = draftObj[field];
    if (typeof val === 'string') {
      return val.trim();
    }
    return '';
  });

  const s = (settings && typeof settings === 'object') ? settings : {};
  const sig = s.signatureState;
  const transform = (sig && typeof sig === 'object')
    ? {
        x: Number(sig.x) || 11.9,
        y: Number(sig.y) || 96.0,
        scale: Math.max(0.15, Math.min(4.5, Number(sig.scale) || 0.5)),
        rot: Number(sig.rot) || 0
      }
    : { x: 11.9, y: 96.0, scale: 0.5, rot: 0 };

  return {
    format: DINLETTER_FORMAT,
    schema_version: Constants.SCHEMA_VERSION,
    generator: 'DIN-BriefNEO',
    generated_at: generatedAt,
    _meta: {
      schema: [...DOCUMENT_SCHEMA_V5]
    },
    layout: {
      form: (s.layout === 'form-a' || s.layout === 'form-b') ? s.layout : 'form-b',
      copyfit: typeof s.copyfit === 'boolean' ? s.copyfit : false,
      marks: typeof s.marks === 'boolean' ? s.marks : true
    },
    fields,
    signature: { transform }
  };
}

/**
 * Validiert und parsed ein DIN-Brief-JSON-Dokument (ausschließlich Schema v4 Tuple-Format).
 * Frühere Schemata werden nicht mehr unterstützt. PURE — unit-testbar.
 * @param {string | object} input
 * @returns {DinLetterParseResult}
 */
export function parseDinLetterPayload(input) {
  /** @type {any} */
  let data;
  if (typeof input === 'object' && input !== null) {
    data = input;
  } else if (typeof input === 'string') {
    const text = input.replace(/^\uFEFF/, '');
    try {
      data = JSON.parse(text);
    } catch {
      return { ok: false, reason: 'Keine gültige JSON-Datei.', error: 'Keine gültige JSON-Datei.' };
    }
  } else {
    return { ok: false, reason: 'Ungültiges Argument (String oder Objekt erwartet).', error: 'Ungültiges Argument (String oder Objekt erwartet).' };
  }

  if (!data || typeof data !== 'object' || Array.isArray(data)) {
    return { ok: false, reason: 'Keine gültige DIN-Brief-Datei (.json, Objekt erwartet).', error: 'Keine gültige DIN-Brief-Datei (.json, Objekt erwartet).' };
  }
  if (data.format !== DINLETTER_FORMAT) {
    return { ok: false, reason: `Falsches Format — erwartet '${DINLETTER_FORMAT}'.`, error: `Falsches Format — erwartet '${DINLETTER_FORMAT}'.` };
  }
  if (data.schema_version !== 5) {
    return {
      ok: false,
      reason: `Nur Schema v5 wird unterstützt (erhalten: v${data.schema_version}).`,
      error: `Nur Schema v5 wird unterstützt (erhalten: v${data.schema_version}).`
    };
  }
  if (!Array.isArray(data.fields) || !Array.isArray(data._meta?.schema)) {
    return {
      ok: false,
      reason: 'Ungültiges Schema-v4-Dokument: fields (Array) und _meta.schema (Array) sind zwingend erforderlich.',
      error: 'Ungültiges Schema-v4-Dokument: fields (Array) und _meta.schema (Array) sind zwingend erforderlich.'
    };
  }

  const schema = data._meta.schema;
  const rawFields = data.fields;
  const fields = /** @type {Record<string, string | boolean>} */ (Object.create(null));

  for (let i = 0; i < schema.length; i++) {
    const key = schema[i];
    if (typeof key !== 'string' || key === 'constructor' || key === 'prototype' || key === '__proto__' || !/^[A-Za-z][\w-]*$/.test(key)) {
      return { ok: false, reason: `Ungültiger Feldname '${key}'.`, error: `Ungültiger Feldname '${key}'.` };
    }
    const val = rawFields[i];
    const istToggle = key.startsWith('toggle-');
    if (istToggle) {
      if (val !== undefined && typeof val !== 'boolean') {
        return { ok: false, reason: `Ungültiger Feldtyp bei '${key}'.`, error: `Ungültiger Feldtyp bei '${key}'.` };
      }
      fields[key] = val === true;
    } else {
      if (val !== undefined && val !== null && typeof val !== 'string') {
        return { ok: false, reason: `Ungültiger Feldtyp bei '${key}'.`, error: `Ungültiger Feldtyp bei '${key}'.` };
      }
      fields[key] = typeof val === 'string' ? val : '';
    }
  }

  /* Kanonische Reihenfolge prüfen (P4, 2026-11-10): Ohne diese Prüfung würde
   * eine Datei mit vertauschten Feldnamen importiert und die Werte still
   * falsch zugeordnet. Die Form-Prüfung oben bleibt zuerst, damit ein kaputter
   * Feldname die spezifischere Meldung bekommt. */
  if (schema.length !== DOCUMENT_SCHEMA_V5.length) {
    return {
      ok: false,
      reason: `_meta.schema hat ${schema.length} Felder, erwartet ${DOCUMENT_SCHEMA_V5.length}.`,
      error: `_meta.schema hat ${schema.length} Felder, erwartet ${DOCUMENT_SCHEMA_V5.length}.`
    };
  }
  for (let i = 0; i < DOCUMENT_SCHEMA_V5.length; i++) {
    if (schema[i] !== DOCUMENT_SCHEMA_V5[i]) {
      return {
        ok: false,
        reason: `_meta.schema[${i}] ist '${schema[i]}', erwartet '${DOCUMENT_SCHEMA_V5[i]}'.`,
        error: `_meta.schema[${i}] ist '${schema[i]}', erwartet '${DOCUMENT_SCHEMA_V5[i]}'.`
      };
    }
  }

  /** @type {Partial<DinLetterLayout>} */
  const layout = Object.create(null);
  if (data.layout !== undefined) {
    const raw = data.layout;
    if (!raw || typeof raw !== 'object' || Array.isArray(raw)) {
      return { ok: false, reason: 'Ungültiges layout-Objekt in der Datei.', error: 'Ungültiges layout-Objekt in der Datei.' };
    }
    if (raw.form !== undefined) {
      if (raw.form !== 'form-a' && raw.form !== 'form-b') {
        return { ok: false, reason: 'Ungültiger layout.form-Wert (form-a oder form-b erwartet).', error: 'Ungültiger layout.form-Wert (form-a oder form-b erwartet).' };
      }
      layout.form = raw.form;
    }
    if (raw.copyfit !== undefined) {
      if (typeof raw.copyfit !== 'boolean') return { ok: false, reason: 'Ungültiger layout.copyfit-Wert (Boolean erwartet).', error: 'Ungültiger layout.copyfit-Wert (Boolean erwartet).' };
      layout.copyfit = raw.copyfit;
    }
    if (raw.marks !== undefined) {
      if (typeof raw.marks !== 'boolean') return { ok: false, reason: 'Ungültiger layout.marks-Wert (Boolean erwartet).', error: 'Ungültiger layout.marks-Wert (Boolean erwartet).' };
      layout.marks = raw.marks;
    }
  }

  /** @type {{ transform: DinLetterSignatureTransform } | undefined} */
  let signature;
  if (data.signature !== undefined && data.signature !== null) {
    if (typeof data.signature !== 'object' || Array.isArray(data.signature)) {
      return { ok: false, reason: 'Ungültiges signature-Objekt in der Datei.', error: 'Ungültiges signature-Objekt in der Datei.' };
    }
    const t = data.signature.transform;
    if (t !== undefined && t !== null) {
      if (typeof t !== 'object' || Array.isArray(t)) {
        return { ok: false, reason: 'Ungültige Signatur-Transformation.', error: 'Ungültige Signatur-Transformation.' };
      }
      if (
        typeof t.x !== 'number' || !Number.isFinite(t.x) ||
        typeof t.y !== 'number' || !Number.isFinite(t.y) ||
        typeof t.scale !== 'number' || !Number.isFinite(t.scale) ||
        typeof t.rot !== 'number' || !Number.isFinite(t.rot)
      ) {
        return { ok: false, reason: 'Ungültige Koordinaten in signature.transform.', error: 'Ungültige Koordinaten in signature.transform.' };
      }
      signature = {
        transform: {
          x: t.x,
          y: t.y,
          scale: Math.max(0.15, Math.min(4.5, t.scale)),
          rot: t.rot
        }
      };
    }
  }

  return {
    ok: true,
    fields,
    layout: Object.keys(layout).length > 0 ? layout : undefined,
    signature,
    schemaVersion: Constants.SCHEMA_VERSION
  };
}

/**
 * Exportiert das DIN-Brief-Payload über die File System Access API (showSaveFilePicker)
 * oder fällt auf einen Standard-Blob-Download über ein <a>-Element zurück.
 * @param {DinLetterPayload} payload
 * @param {string} [fileName]
 * @returns {Promise<boolean>}
 */
export async function exportDocumentToFile(payload, fileName = buildLetterFileName()) {
  const jsonContent = JSON.stringify(payload, null, 2);
  const fullName = fileName.endsWith('.json') ? fileName : `${fileName}.json`;

  const savePicker = typeof window !== 'undefined' ? /** @type {any} */ (window).showSaveFilePicker : null;
  if (typeof savePicker === 'function') {
    try {
      const handle = await savePicker.call(window, {
        suggestedName: fullName,
        types: [{
          description: 'DIN-Brief Dokument',
          accept: {
            'application/json': ['.json', '.dinletter']
          }
        }]
      });
      const writable = await handle.createWritable();
      await writable.write(jsonContent);
      await writable.close();
      notify(Constants.TOASTS.DRAFT.EXPORT_SAVED, 'info');
      return true;
    } catch (err) {
      if (err instanceof DOMException && err.name === 'AbortError') {
        return false;
      }
      console.warn('[Export] File System Access API fehlgeschlagen, Fallback auf Download:', err);
    }
  }

  // Fallback: Blob-Download über temporäres <a>-Element
  try {
    const blob = new Blob([jsonContent], { type: 'application/json' });
    const url = URL.createObjectURL(blob);
    const anchor = document.createElement('a');
    anchor.href = url;
    anchor.download = fullName;
    anchor.click();
    setTimeout(() => URL.revokeObjectURL(url), 1000);
    notify(Constants.TOASTS.DRAFT.EXPORT_DOWNLOADED, 'info');
    return true;
  } catch (err) {
    console.error('[Export] Download fehlgeschlagen:', err);
    notify(Constants.TOASTS.DRAFT.EXPORT_ERROR, 'error');
    return false;
  }
}

/** @type {(() => Record<string, string | boolean>) | null} */
let registeredDraftCollector = null;

/**
 * Registriert den Collector für den aktuellen DOM-Draft.
 * @param {(() => Record<string, string | boolean>) | null} collector
 */
export function registerDraftCollector(collector) {
  registeredDraftCollector = collector;
}

/**
 * Erzeugt das vollständige DIN-Letter-Payload (Objekt), wie es vom Export-Button
 * und vom Live-JSON-Debug-Panel verwendet wird. PURE / SSoT.
 * @param {(() => Record<string, string | boolean>) | null} [draftCollector]
 * @returns {DinLetterPayload}
 */
export function buildDINLetterJSON(draftCollector = null) {
  const collector = draftCollector || registeredDraftCollector;
  let draft;
  try {
    draft = typeof collector === 'function' ? collector() : loadJSON(STORAGE_KEYS.draft, {});
  } catch {
    draft = loadJSON(STORAGE_KEYS.draft, {});
  }
  const settings = loadJSON(STORAGE_KEYS.settings, null);
  return buildDinLetterPayload(draft, settings);
}

/**
 * Wendet einen geprüften Import an und persistiert ihn im localStorage.
 * Bricht bei Speicherfehlern (saveJSON -> false) sofort ab, ohne reload auszulösen (B55).
 * @param {any} pendingImport
 * @param {{ reloadFn?: () => void }} [options]
 * @returns {boolean} true bei Erfolg, false bei Speicherfehler
 */
export function applyImportedData(pendingImport, { reloadFn = () => location.reload() } = {}) {
  if (!pendingImport?.ok) return false;
  try {
    const draftGespeichert = saveJSON(STORAGE_KEYS.draft, pendingImport.fields);
    const currentSettings = loadJSON(STORAGE_KEYS.settings, DEFAULT_SETTINGS) || DEFAULT_SETTINGS;
    /** @type {Record<string, any>} */
    const nextSettings = { ...currentSettings };
    const layout = pendingImport.layout;
    if (layout) {
      if (layout.form !== undefined) nextSettings.layout = layout.form;
      if (layout.copyfit !== undefined) nextSettings.copyfit = layout.copyfit;
      if (layout.marks !== undefined) nextSettings.marks = layout.marks;
    }
    if (pendingImport.signature?.transform) {
      nextSettings.signatureState = { ...pendingImport.signature.transform };
    }
    const settingsGespeichert = saveJSON(STORAGE_KEYS.settings, nextSettings);
    // saveJSON gibt bei vollem Speicher false zurück statt zu werfen (B55).
    // Ohne diese Prüfung meldet der Dialog Erfolg und der Reload verwirft den Import.
    if (!draftGespeichert || !settingsGespeichert) {
      notify(Constants.TOASTS.DRAFT.IMPORT_STORAGE_ERROR, 'error');
      return false;
    }
    if (typeof reloadFn === 'function') {
      reloadFn();
    }
    return true;
  } catch {
    notify(Constants.TOASTS.DRAFT.IMPORT_STORAGE_ERROR, 'error');
    return false;
  }
}

/**
 * Verdrahtet Export/Import-UI (Sidebar-Buttons, Datei-Input, Confirm-Dialog, Keyboard-Shortcut).
 * @param {{ collectDraft?: () => Record<string, string | boolean> }} [options]
 */
export function initImportExport({ collectDraft } = {}) {
  if (typeof collectDraft === 'function') {
    registerDraftCollector(collectDraft);
  }

  const exportBtn = /** @type {HTMLButtonElement | null} */ (document.getElementById('btn-export-dinletter'));
  const copyBtn = /** @type {HTMLButtonElement | null} */ (document.getElementById('btn-copy-dinletter'));
  const importBtn = /** @type {HTMLButtonElement | null} */ (document.getElementById('btn-import-dinletter'));
  const importInput = /** @type {HTMLInputElement | null} */ (document.getElementById('dinletter-uploader'));
  const importDialog = /** @type {HTMLDialogElement | null} */ (document.getElementById('import-dialog'));
  const importFileName = document.getElementById('import-file-name');
  if (!exportBtn || !importBtn || !importInput || !importDialog) return;

  const handleExport = async () => {
    // Kein DOCUMENT_SAVE_REQUESTED: das Payload entsteht direkt aus dem DOM
    // (collectDraft). Der Request löste nur einen zweiten Speichervorgang aus (B47).
    const payload = buildDINLetterJSON(collectDraft);
    await exportDocumentToFile(payload);
  };

  exportBtn.addEventListener('click', handleExport);

  if (copyBtn) {
    copyBtn.addEventListener('click', async () => {
      try {
        const payload = buildDINLetterJSON(collectDraft);
        const jsonContent = JSON.stringify(payload, null, 2);
        await navigator.clipboard.writeText(jsonContent);
        notify(Constants.TOASTS.DRAFT.EXPORT_CLIPBOARD, 'info');
      } catch (err) {
        console.error('[Export] Kopieren in Zwischenablage fehlgeschlagen:', err);
        notify(Constants.TOASTS.DRAFT.EXPORT_ERROR, 'error');
      }
    });
  }

  // Globaler Tastatur-Shortcut: Ctrl + S / Cmd + S triggert exportDocumentToFile
  window.addEventListener('keydown', (e) => {
    if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === 's') {
      e.preventDefault();
      handleExport();
    }
  });

  importBtn.addEventListener('click', () => {
    importInput.value = '';
    importInput.click();
  });

  /** @type {DinLetterParseResult | null} */
  let pendingImport = null;

  importInput.addEventListener('change', async () => {
    const file = importInput.files?.[0];
    if (!file) return;
    if (file.size > 1024 * 1024) {
      notify(Constants.TOASTS.DRAFT.IMPORT_SIZE_ERROR, 'error');
      return;
    }
    const result = parseDinLetterPayload(await file.text());
    if (!result.ok) {
      notify(Constants.TOASTS.DRAFT.IMPORT_REJECTED(result.reason), 'error');
      return;
    }
    pendingImport = result;
    if (importFileName) {
      let info = file.name;
      const details = [];
      if (result.layout?.form) {
        details.push(`Layout: ${result.layout.form === 'form-a' ? 'Form A' : 'Form B'}`);
      }
      if (result.signature?.transform) {
        details.push('Signatur-Position');
      }
      if (details.length > 0) {
        info += ` (${details.join(', ')})`;
      }
      Field.setValue(importFileName, info);
    }
    importDialog.showModal();
  });

  importDialog.addEventListener('close', () => {
    const confirmed = importDialog.returnValue === 'confirm';
    importDialog.returnValue = '';
    if (!confirmed || !pendingImport?.ok) {
      pendingImport = null;
      return;
    }
    const toApply = pendingImport;
    pendingImport = null;
    applyImportedData(toApply);
  });
}
